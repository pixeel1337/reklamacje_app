router.get("/:name/:countryCode/:birthYear/offspring", async (req, res) => {
    try {
        const { name, countryCode, birthYear } = req.params;
        
        // 1. Wyciągamy opcjonalny parametr z req.query (np. ?bornAfter=2020)
        const { bornAfter } = req.query;
        
        const country = await Country.findOne({ code: countryCode.toUpperCase() });
        if(!country) {
            return res.status(404).json({ message: "Niepoprawny kod kraju!" });
        }

        const parent = await Horse.findOne({ name, birthCountry: country._id, birthYear: parseInt(birthYear) });
        
        if (!parent) {
            return res.status(404).json({ message: "Nie znaleziono konia-rodzica o podanych danych!" });
        }

        // 2. Budujemy podstawowy filtr: koń musi być potomkiem (ojciec LUB matka)
        let queryFilter = {
            $or: [
                { father: parent._id },
                { mother: parent._id }
            ]
        };

        // 3. Dynamiczny krok: Jeśli użytkownik podał ?bornAfter, dorzucamy operator $gt do filtra
        if (bornAfter) {
            queryFilter.birthYear = { $gt: parseInt(bornAfter) };
        }

        // 4. Odpytujemy bazę danych przy użyciu przygotowanego dynamicznie obiektu queryFilter
        const offspring = await Horse.find(queryFilter)
        .populate("breeder")
        .populate("birthCountry");

        return res.status(200).json({
            parent: {
                name: parent.name,
                birthYear: parent.birthYear,
                gender: parent.gender
            },
            count: offspring.length,
            offspring: offspring
        });

    } catch (err) {
        return res.status(500).json({ error: "Błąd podczas pobierania potomstwa: " + err.message });
    }
});





router.get("/", async (req, res) => {
    try {
        // 1. Pobieramy parametry stronicowania z req.query (z bezpiecznymi wartościami domyślnymi)
        const page = parseInt(req.query.page) || 1;    // Jeśli nie podano, startujemy od 1 strony
        const limit = parseInt(req.query.limit) || 10; // Jeśli nie podano, bierzemy 10 koni na stronę
        
        // Obliczamy przesunięcie (ile koni z początku bazy trzeba pominąć)
        const skip = (page - 1) * limit;

        // 2. Liczymy, ile W SUMIE koni znajduje się w całej bazie danych
        const totalHorses = await Horse.countDocuments({});

        // 3. Pobieramy tylko wybraną paczkę (stronę) koni
        const horses = await Horse.find({})
            .populate("breeder")
            .populate("birthCountry")
            .populate("father")
            .populate("mother")
            .sort({ birthYear: -1 }) // Sortujemy od najmłodszych (najwyższe roczniki na górze)
            .skip(skip)              // Przeskakujemy konie z poprzednich stron
            .limit(limit);           // Ograniczamy wynik do zadeklarowanego limitu

        // 4. Zwracamy konie oraz pełne metadane stronicowania
        return res.status(200).json({
            pagination: {
                totalResults: totalHorses,                  // Łączna liczba wszystkich koni w bazie
                totalPages: Math.ceil(totalHorses / limit), // Ile stron ma cała baza przy tym limicie
                currentPage: page,                          // Na której stronie aktualnie znajduje się użytkownik
                resultsPerPage: limit                       // Ile koni jest zwracanych na jedną stronę
            },
            horses: horses // Tablica z wyciętym fragmentem danych (maksymalnie 'limit' obiektów)
        });

    } catch(err) {
        return res.status(500).json({ error: "Błąd bazy danych: " + err.message });
    }
});
